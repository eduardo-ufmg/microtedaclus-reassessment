\chapter{Background and Related Work}
\label{chap:background_relatedwork}

This chapter presents the eccentricity and the typicality of the TEDA framework (\cref{sec:teda}), the formulation of \MicroTEDAClus by Maia et al.\ (\cref{sec:original_formulation}), its update by Galindo (\cref{sec:updated_formulation}), and the reimplementation of the model in \codename{C} by Cardoso (\cref{sec:c_implementation}). After each formulation, we discuss its issues, and we point to where the following chapters examine them.

\section{Typicality and Eccentricity-Based Data Analytics}
\label{sec:teda}

Angelov~\cite{angelov2014outside} introduces the TEDA framework as an alternative to probability theory for data in non-deterministic settings, built on two metrics, the eccentricity and the typicality. Let

\begin{equation}
    C = \{ \sum_{i=1}^{k} \pi^k_i > 0,\quad j > 1,\quad k > 2 \}
\end{equation}

be the conditions under which the metrics of a sample $j$ are computed at moment $k$, where $\pi^k_j$ is the sum of the distances from sample $j$ to all the samples seen up to $k$. The eccentricity, which measures how much a sample deviates from the others, and the typicality, which measures how well it fits among them, are

\begin{equation}
    \xi^k_j = \frac{2\pi^k_j}{\sum_{i=1}^{k} \pi^k_i}, \quad C
    \label{eq:eccentricity_definition}
\end{equation}

\begin{equation}
    \tau^k_j = 1 - \xi^k_j, \quad C
    \label{eq:typicality_definition}
\end{equation}

Both are bounded, $0 < \xi^k_j < 1$ and $0 < \tau^k_j < 1$, and their normalized forms are

\begin{equation}
    \zeta^k_j = \frac{\xi^k_j}{2}, \quad \sum_{i=1}^{k} \zeta^k_i = 1, \quad 0 < \zeta^k_j < \frac{1}{2}, \quad C
    \label{eq:normalized_eccentricity}
\end{equation}

\begin{equation}
    t^k_j = \frac{\tau^k_j}{k - 2}, \quad \sum_{i=1}^{k} t^k_i = 1, \quad 0 < t^k_j < \frac{1}{k - 2}, \quad C
    \label{eq:normalized_typicality}
\end{equation}

A sample is anomalous if $\zeta^k_j > 1/k$ and typical if $t^k_j > 1/k$. For the squared Euclidean distance, the sum of the distances can be updated recursively from the mean and the mean squared norm of the samples,

\begin{equation}
\begin{gathered}
    \mu^k = \frac{k - 1}{k} \mu^{k-1} + \frac{1}{k} x_k, \qquad
    X^k = \frac{k - 1}{k} X^{k-1} + \frac{1}{k} \lVert x_k \rVert^2,
\end{gathered} \label{eq:recursive_mu_X}
\end{equation}

since

\begin{equation}
    \pi^k_j = k \left(\lVert x_j - \mu^k \rVert^2 + X^k - \lVert \mu^k \rVert^2\right), \qquad
    \sum_{i=1}^{k} \pi^k_i = \sum_{i=1}^{k-1} \pi^{k-1}_i + 2 \pi^k_k.
    \label{eq:recursive_sum_pi}
\end{equation}

Angelov also states that \labeltext[Angelov's statement about the use of other proximity functions]{the metrics suit other proximity functions, among them the cosine similarity and the Mahalanobis distance}{info:mahalanobis_as_teda_metric}~\cite{mahalanobis1936generalized}.

\section{Original MicroTEDAClus}
\label{sec:original_formulation}

Maia et al.~\cite{maia2020evolvclus} use the eccentricity and the typicality over the Euclidean distance to create and update microclusters, by detecting outliers and typical samples, and to merge the microclusters into macroclusters. They write the eccentricity in the recursive form

\begin{equation}
    \xi(x_k) = \frac{1}{k} + \frac{\left( \mu_k - x_k \right)^T \left( \mu_k - x_k \right)}{k \sigma^2_k}, \label{eq:recursive_eccentricity}
\end{equation}

with

\begin{gather}
    \mu_k = \frac{k - 1}{k} \mu_{k-1} + \frac{x_k}{k}, \qquad
    \sigma^2_k = \frac{k - 1}{k} \sigma^2_{k-1} + \frac{1}{k - 1} \lVert x_k - \mu_k \rVert^2,
\end{gather}

and consider a sample an outlier if

\begin{equation}
    \zeta(x_k) > \frac{m^2 + 1}{2k}, \quad m > 0, \label{eq:angelovs_outlier}
\end{equation}

where $m$ is a function of the number of samples chosen by the authors,

\begin{equation}
    m(k) = \frac{3}{1 + e^{-0.007(k - 100)}}.
    \label{eq:m_function}
\end{equation}

A microcluster $i$ at moment $k$ is described by:

\begin{itemize}
    \item $S^i_k$, its number of samples;
    \item $\mu^i_k$ and $(\sigma^i_k)^2$, the mean and the variance of its samples;
    \item $\xi^i(x_k)$ and $\zeta^i(x_k)$, the eccentricity and the normalized eccentricity of sample $x_k$ to it;
    \item $\tau^i(x_k)$ and $t^i(x_k)$, the typicality and the normalized typicality of $x_k$ to it;
    \item $D^i_k = 1 / \zeta^i(x_k)$, \labeltext[order-dependent parameter handled as a characteristic of the microcluster]{its density}{critique:density_as_microcluster_characteristic};
    \item $m^i_k(S^i_k)$, its threshold function.
\end{itemize}

The metrics apply to each microcluster as to the whole data. For $S^i_k > 2$, a sample $x_k$ is an outlier to microcluster $i$ if

\begin{equation}
    \zeta^i(x_k) > \frac{m^i_k(S^i_k)^2 + 1}{2S^i_k}, \label{eq:outlier_condition}
\end{equation}

and, for $S^i_k = 2$, if

\begin{equation}
    \left( \zeta^i(x_k) > \frac{m^i_k(2)^2 + 1}{4} \right) \land \left( \left( \sigma^i_k \right)^2 < r_0 \right), \label{critique:incorrect_condition_logic}
\end{equation}

where $r_0$ is a hyperparameter, the variance limit. Every microcluster to which $x_k$ is not an outlier is updated by

\begin{equation}
\begin{gathered}
    S^i_k = S^i_{k-1} + 1, \qquad
    \mu^i_k = \frac{S^i_k - 1}{S^i_k} \mu^i_{k-1} + \frac{x_k}{S^i_k}, \\
    \left(\sigma^i_k\right)^2 = \frac{S^i_k - 1}{S^i_k} \left(\sigma^i_{k-1}\right)^2 + \frac{1}{S^i_k - 1} \left( \frac{2\lVert x_k - \mu^i_k \rVert }{d} \right)^2, \\
    \xi^i(x_k) = \frac{1}{S^i_k} + \frac{2\left( \mu^i_k - x_k \right)^T \left( \mu^i_k - x_k \right)}{S^i_k \left( \sigma^i_k \right)^2d},
\end{gathered} \label{critique:dimensionality_penalty}
\end{equation}

where $d$ is the dimension of the data, and a sample that is an outlier to every microcluster creates a new one with

\begin{equation}
    S^i_k = 1, \quad \mu^i_k = x_k, \quad (\sigma^i_k)^2 = 0, \label{eq:initial_microcluster_parameters}
\end{equation}

and the first sample of the stream creates the first microcluster in the same way.

The macroclusters are the connected components of the graph whose vertices are the microclusters and whose edges join the microclusters that satisfy

\begin{equation}
    \text{dist}(\mu^i_k, \mu^j_k) < 2 \left( \sigma^i_k + \sigma^j_k \right), \quad i \neq j. \label{eq:microcluster_intersection_condition}
\end{equation}

\labeltext[criterion to prevent microcluster collapse]{To keep overlapping microclusters from collapsing into a single macrocluster, a microcluster $m^j_l$ takes part in the macrocluster $\mathcal{M}_j = \{ m^j_1, \dots, m^j_{N_j} \}$ only if it is active, that is, if}{info:microcluster_activation_criterion}

\begin{equation}
    D^l_k \ge \text{mean}\left(D^l_k\right), \quad l = 1,\dots,N_j. \label{critique:use_of_unstable_density}
\end{equation}

Each active microcluster weighs in its macrocluster by its share of the density of the active microclusters,

\begin{equation}
    w^l_k = \frac{D^l_k}{\sum_{l\in\mathcal{M}_j} D^l_k},
    \label{eq:macrocluster_density}
\end{equation}

the typicality of a sample $x_k$ to a macrocluster is the weighted sum of its typicalities to the active microclusters,

\begin{equation}
    \mathcal{T}_j(x_k) = \sum_{l\in\mathcal{M}_j} w^l_k t^l_k(x_k), \label{eq:macrocluster_typicality}
\end{equation}

and the sample is assigned to the macrocluster of highest typicality,

\begin{equation}
    j = \arg\max_j \mathcal{T}_j(x_k),
    \label{eq:macrocluster_prediction}
\end{equation}

The formulation defines no decay of the microclusters, although the public implementation has one (\cref{sec:model_implementation}).

\subsection{Issues of the Original Formulation}

\paragraph{Order-dependent density.}
\label{subsubsec:order_dependent_characteristics}
The eccentricity and the typicality relate a sample to a microcluster, so the density $D^i_k = 1 / \zeta^i(x_k)$ relates them too, and should not be treated as a property of the microcluster. A microcluster with a small variance is dense for any standard clustering algorithm~\cite{ester1996density,aggarwal2003framework,cao2006density}, but if the current sample is far from its mean, $\zeta^i(x_k)$ is large and $D^i_k$ small, and the microcluster is less likely to be activated. Conversely, a microcluster with a large variance and a sample close to its mean has a large $D^i_k$ and is more likely to be activated, even if it is not dense at all. \Cref{sec:mi_corrections} tests two densities that do not depend on the current sample, and \cref{subsec:mi_density_revisited} tests both again on \MicroTEDAClusPlus.

\paragraph{Outlier condition for microclusters with two samples.}
\label{subsubsec:incorrect_condition_logic}
For $S^i_k = 2$, the expressions of \cref{sec:teda} give

\begin{gather}
    \mu_2 = \frac{1}{2} \left( x_1 + x_2 \right), \qquad
    \sigma^2_2 = \lVert x_2 - \mu_2 \rVert^2, \qquad
    \xi(x_2) = 1, \qquad
    \zeta(x_2) = 0.5,
\end{gather}

and, because $m \ge 1$,

\begin{equation}
    \frac{m^i_k(2)^2 + 1}{4} \ge 0.5.
\end{equation}

The first term of \cref{critique:incorrect_condition_logic} is therefore never satisfied, and neither is the conjunction: no sample can be an outlier to a microcluster with two samples. With the scaled expressions of \cref{critique:dimensionality_penalty}, the normalized eccentricity of the second sample becomes $1/4 + d/8$, so the first term is never satisfied with one or two dimensions and always satisfied with three or more, where the condition reduces to $\left( \sigma^i_k \right)^2 < r_0$, which holds for the samples close to the mean. Either way, the condition does not detect outliers. Replacing it by

\begin{equation}
    \left( \sigma^i_k \right)^2 < r_0
    \label{eq:corrected_condition_logic}
\end{equation}

would keep the inversion, so the condition should be

\begin{equation}
    \left( \sigma^i_k \right)^2 > r_0.
    \label{eq:corrected_condition_logic_for_outliers}
\end{equation}

Cardoso~\cite{cardoso2022microtedaclus} reached the same conclusion earlier, proving that the eccentricity of the second sample is always one, and detects the outliers of microclusters with fewer than three samples by \cref{eq:corrected_condition_logic_for_outliers} alone, as the public implementation does (\cref{sec:model_implementation}).

\paragraph{Dimensionality penalty.}
\label{subsubsec:dimensionality_penalty}
\Cref{critique:dimensionality_penalty} scales the variance and the eccentricity of a microcluster by the dimension $d$. No paper that introduces the TEDA metrics suggests such a scaling~\cite{angelov2014outside,angelov2014anomaly}, the original work does not justify it, and it lets the normalized eccentricity exceed its bound of one half. \Cref{sec:reassessment_results} traces the collapse of \MicroTEDAClus in high dimensions to it, and \cref{sec:mi_corrections} tests its removal.

\section{Updated MicroTEDAClus}
\label{sec:updated_formulation}

In the PhD thesis \textit{Mineração de Dados Aplicada a Problemas de Last-Mile na Logística}, Galindo~\cite{galindo2021lastmile} updates \MicroTEDAClus to apply it to the last-mile problem in logistics; we describe only the changes. The condition for microclusters with two samples becomes

\begin{equation}
    \left( \zeta^i_2(x_2)  > \frac{m^i(2)^2+1}{4} \right) \land \left( \left( \sigma^i_k \right)^2 > r_0 \right),
    \label{eq:updated_condition_logic_for_two_samples}
\end{equation}

which corrects the direction of the comparison with $r_0$ but keeps the conjunction.

An explicit decay is introduced. Each microcluster has a vitality $v^i_k$, set to one when it is created and lowered by

\begin{equation}
    v^i_k = v^i_{k-1} - \rho
    \label{eq:decay_logic}
\end{equation}

whenever the sample $x_k$ is an outlier to it, where $\rho = 1/\text{decay}$ and \textit{decay} is a hyperparameter. A microcluster is removed when $v^i_k \le 0$. \Cref{sec:mi_improvements} tests this rule as the candidate \variant{recency}.

To suit the data of the problem, the Euclidean distance is replaced by the cosine similarity, which Angelov~\cite{angelov2014outside} accepts for the TEDA metrics. The outlier condition becomes

\begin{equation}
    \left(x_k - \mu^i_k\right)^2 > \left(m^i_k\right)^2\left(\sigma^i_k\right)^2, \label{critique:imports_euclidean_condition_from_angelov}
\end{equation}

and the update of a microcluster becomes

\begin{equation}
\begin{gathered}
    S^i_k = S^i_{k-1} + 1, \qquad
    \mu^i_k = \frac{S^i_k - 1}{S^i_k} \mu^i_{k-1} + \frac{1}{S^i_k} \frac{x_k}{\lVert x_k \rVert}, \quad \lVert x_k \rVert \neq 0, \\
    \left(\sigma^i_k\right)^2 = \frac{S^i_k - 1}{S^i_k} \left(\sigma^i_{k-1}\right)^2 + \frac{1}{S^i_k - 1} \left( \frac{2\lVert x_k - \mu^i_k \rVert}{d} \right)^2, \label{critique:euclidean_variance} \\
    \xi^i(x_k) = \frac{1 - \mu^i_k\frac{x_k}{\lVert x_k \rVert}}{1 + k\left(\mu^i_k\right)^2}, \quad \lVert x_k \rVert \neq 0.
\end{gathered}
\end{equation}

The rest of the algorithm, including the formation of the macroclusters, is unchanged.

\subsection{Issues of the Updated Formulation}

\paragraph{Issues that remain.} The order-dependent density and the dimensionality penalty remain, and the condition for microclusters with two samples keeps its conjunction.

\paragraph{An outlier condition derived for another distance.} Angelov~\cite{angelov2014anomaly} derives \cref{eq:angelovs_outlier} from \cref{critique:imports_euclidean_condition_from_angelov} for the Euclidean distance, and states that a similar condition can be derived for the cosine similarity without deriving it. The updated model uses the Euclidean condition with the cosine similarity without justification.

\paragraph{A Euclidean variance with the cosine similarity.} The eccentricity is adapted to the cosine similarity, but the variance of \cref{critique:euclidean_variance} is still the Euclidean one, and nothing shows that it is compatible with the cosine similarity.

\section{A \texttt{C} Implementation of MicroTEDAClus}
\label{sec:c_implementation}

In the undergraduate thesis \textit{Implementação do algoritmo MicroTEDAclus para classificação de dados}, Cardoso~\cite{cardoso2022microtedaclus} reimplements the original \MicroTEDAClus in \codename{C}, a language chosen for its speed and for its use in embedded platforms. The thesis documents the TEDA expressions that the model uses, with proofs and unit tests, and proposes an alternative to the intersection condition of \cref{eq:microcluster_intersection_condition}, which compares the distance between the means of two microclusters with the sum of their radii $m^i_k(S^i_k)\,\sigma^i_k$, scaled by a factor $\phi$:

\begin{equation}
    \text{dist}(\mu^i_k, \mu^j_k) < \phi \left( m^i_k(S^i_k)\,\sigma^i_k + m^j_k(S^j_k)\,\sigma^j_k \right), \quad \phi \ge 1, \quad i \neq j.
    \label{eq:cardoso_intersection_condition}
\end{equation}

Its experiments evaluate the clustering quality by the adjusted Rand index on seven static datasets of two dimensions. The model learns each whole dataset before labeling it, in random orders of the samples and with hyperparameters taken from a grid, and the thesis reports the lowest and the highest index and the highest mean index over the configurations~\cite[Section 4.3]{cardoso2022microtedaclus}. It concludes that the quality depends strongly on $r_0$ and $\phi$, and that \cref{eq:cardoso_intersection_condition} is comparable to \cref{eq:microcluster_intersection_condition}. The reimplementation runs much faster than the \codename{Python} implementation of the original work, measured over each whole program on a single machine and without repetitions~\cite[Section 4.4]{cardoso2022microtedaclus}.

\subsection{Issues of the Evaluation of the C Implementation}

The highest indices belong to the configurations of the grid that best fit the labels of the evaluated data, and are therefore optimistic estimates of the quality that the model reaches when its hyperparameters are chosen without those labels. The lowest index is no better than chance in every dataset~\cite[Tables 4.1 and 4.2]{cardoso2022microtedaclus}, and the comparison between the two intersection conditions rests on the best cases, without a statistical test. The model is compared with no benchmark model, and none of the datasets has concept drift, which the thesis expects the model to handle because the original work reports that it does~\cite[Section 4.2]{cardoso2022microtedaclus}.
